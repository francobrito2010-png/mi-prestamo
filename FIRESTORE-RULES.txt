rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn()   { return request.auth != null && request.auth.token.email != null; }
    function accExists()   { return exists(/databases/$(database)/documents/meta/access); }
    function acc()         { return get(/databases/$(database)/documents/meta/access).data; }
    function myEmail()     { return request.auth.token.email.lower(); }
    function isOwner()     { return signedIn() && accExists() && myEmail() == acc().owner.lower(); }
    function isMember()    { return signedIn() && accExists() && ( isOwner() || (myEmail() in acc().members) ); }

    // Documento de control de acceso (dueño + personas autorizadas + código admin)
    match /meta/access {
      allow read:   if signedIn();
      // Creación inicial: solo si aún no existe y el creador se pone como dueño y miembro
      allow create: if signedIn()
                    && !accExists()
                    && request.resource.data.owner.lower() == myEmail()
                    && (myEmail() in request.resource.data.members);
      // Solo el dueño actual puede cambiar accesos / código, y sigue siendo dueño
      allow update: if isOwner() && request.resource.data.owner.lower() == myEmail();
      allow delete: if false;
    }

    // Datos del préstamo: cualquier miembro lee, solo el dueño escribe
    match /loan/{doc=**} {
      allow read:  if isMember();
      allow write: if isOwner();
    }

    // Cuotas/pagos: los miembros leen y REGISTRAN; solo el dueño edita/borra
    match /payments/{id} {
      allow read:   if isMember();
      allow create: if isMember()
                    && request.resource.data.amount is number
                    && (!('receipt' in request.resource.data) || request.resource.data.receipt.size() < 900000);
      allow update, delete: if isOwner();
    }

    // Catch-all: bloquear TODO lo demás (dejar SIEMPRE al final)
    match /{document=**} { allow read, write: if false; }
  }
}
